cmake_minimum_required(VERSION 3.13)
set(PICO_BOARD pico_w CACHE STRING "Pico board")
include(pico_sdk_import.cmake)
project(imu_features C CXX ASM)
pico_sdk_init()
add_subdirectory(icm20948)
add_executable(imu_features main.c features.c)
target_include_directories(imu_features PRIVATE icm20948)
target_link_libraries(imu_features pico_stdlib hardware_i2c icm20948 m)
pico_enable_stdio_usb(imu_features 1)
pico_enable_stdio_uart(imu_features 0)
pico_add_extra_outputs(imu_features)
